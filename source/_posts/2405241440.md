# 安装依赖

首先需要安装相关依赖包：

```bash
npm install @nestjs/cache-manager cache-manager
```

# 模块配置

```ts
import { Global, Module } from '@nestjs/common'
import { CacheModule } from '@nestjs/cache-manager'
import { MemoryCacheService } from './memory-cache.service'

@Global()
@Module({
  imports: [
    CacheModule.register({
      ttl: 60 * 60 * 1000,
      max: 1000,
      isGlobal: false,
    }),
  ],
  providers: [MemoryCacheService],
  exports: [CacheModule, MemoryCacheService],
})
export class MemoryCacheModule {}
```

配置参数说明：
- `ttl`: 缓存默认存活时间（毫秒），**v4 版本以秒为单位，v5 版本以毫秒为单位**
- `max`: 缓存中最大允许存储的条目数，超出后使用 LRU 策略淘汰
- `isGlobal`: 是否注册为全局模块

> 💡 如果使用 `@Global()` 装饰器，就不需要再设置 `isGlobal: true`，二者选其一即可。

# 服务封装

```ts
import { Inject, Injectable, Logger } from '@nestjs/common'
import { CACHE_MANAGER } from '@nestjs/cache-manager'
import { Cache } from 'cache-manager'

@Injectable()
export class MemoryCacheService {
  private readonly logger = new Logger(MemoryCacheService.name)

  constructor(@Inject(CACHE_MANAGER) private memoryCache: Cache) {}

  async get<T = any>(cacheKey: string): Promise<T | undefined> {
    return this.memoryCache.get<T>(cacheKey)
  }

  async set<T>(key: string, value: T, cacheTime?: number): Promise<void> {
    await this.memoryCache.set(key, value, cacheTime)
  }

  async del(key: string): Promise<void> {
    await this.memoryCache.del(key)
  }

  async reset(): Promise<void> {
    await this.memoryCache.reset()
  }

  async keys(): Promise<string[]> {
    const store = this.memoryCache.store as any
    return store.keys()
  }
}
```

# 使用示例

```ts
@Injectable()
export class UserService {
  constructor(private readonly memoryCache: MemoryCacheService) {}

  async getUserInfo(userId: number): Promise<UserInfo> {
    const cacheKey = `user:info:${userId}`
    const cached = await this.memoryCache.get<UserInfo>(cacheKey)

    if (cached) {
      return cached
    }

    const userInfo = await this.queryUserFromDb(userId)
    await this.memoryCache.set(cacheKey, userInfo, 5 * 60 * 1000)
    return userInfo
  }

  async clearUserCache(userId: number): Promise<void> {
    await this.memoryCache.del(`user:info:${userId}`)
  }
}
```

# 清空所有缓存

```ts
await this.memoryCache.reset()
```

# 使用场景

内存缓存适合以下场景：
- 缓存数据量不大，但访问频率很高
- 数据允许丢失，重启应用后可以重新生成
- 单实例部署，或每个实例独立缓存即可
- 对响应延迟要求极高，需要毫秒级读取

# 文件缓存 vs 内存缓存对比

| 特性 | 内存缓存 | 文件缓存 |
|------|----------|----------|
| 读取速度 | 极快 | 较快 |
| 数据持久化 | 重启丢失 | 重启保留 |
| 缓存容量 | 受内存限制 | 受磁盘限制 |
| LRU 淘汰 | 支持 | 不支持 |
| 适用场景 | 热点小数据 | 大数据/持久化 |
