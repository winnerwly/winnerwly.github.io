# 安装依赖

首先需要安装相关依赖包：

```bash
npm install @nestjs/cache-manager cache-manager cache-manager-fs-hash
```

# 模块配置

```ts
import { Module } from '@nestjs/common'
import { CacheModule } from '@nestjs/cache-manager'
import * as fsHashStore from 'cache-manager-fs-hash'
import { FileCacheService } from './file-cache.service'

@Module({
  imports: [
    CacheModule.register({
      store: fsHashStore,
      path: 'cache',
      ttl: 60 * 60 * 1000,
    }),
  ],
  providers: [FileCacheService],
  exports: [CacheModule, FileCacheService],
})
export class FileCacheModule {}
```

配置参数说明：
- `store`: 指定使用文件系统存储
- `path`: 缓存文件存储目录
- `ttl`: 默认缓存过期时间（毫秒），cache-manager-fs-hash v5 版本以毫秒为单位

# 服务封装

```ts
import { Inject, Injectable, Logger } from '@nestjs/common'
import { CACHE_MANAGER } from '@nestjs/cache-manager'
import { Cache } from 'cache-manager'

@Injectable()
export class FileCacheService {
  private readonly logger = new Logger(FileCacheService.name)

  constructor(@Inject(CACHE_MANAGER) private fileCache: Cache) {}

  async get<T = any>(cacheKey: string): Promise<T | undefined> {
    return this.fileCache.get<T>(cacheKey)
  }

  async set<T>(key: string, value: T, cacheTime?: number): Promise<void> {
    await this.fileCache.set(key, value, cacheTime)
  }

  async del(key: string): Promise<void> {
    await this.fileCache.del(key)
  }

  async reset(): Promise<void> {
    await this.fileCache.reset()
  }
}
```

# 使用示例

在业务服务中注入并使用：

```ts
@Injectable()
export class DataService {
  constructor(private readonly fileCache: FileCacheService) {}

  async getHeavyData(): Promise<Data> {
    const cacheKey = 'heavy-data:v1'
    const cached = await this.fileCache.get<Data>(cacheKey)
    
    if (cached) {
      return cached
    }

    const data = await this.fetchFromDatabase()
    await this.fileCache.set(cacheKey, data, 24 * 60 * 60 * 1000)
    return data
  }
}
```

# 使用场景

文件缓存适合以下场景：
- 需要持久化缓存数据，重启应用后不丢失
- 缓存数据量较大，不适合全部放在内存中
- 多进程/多实例部署，每个实例独立缓存
- 数据不频繁变更，读取远多于写入
