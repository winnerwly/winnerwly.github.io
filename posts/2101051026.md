## 安装 egg-jwt 插件
```bash
npm install egg-jwt --save

# 或者

yarn add egg-jwt
```
## 添加全局配置
```js
// /app/config/plugin.js
jwt: {
  enable: true,
  package: 'egg-jwt',
}
```
```js
// /app/config/config.default.js
config.jwt = {
  secret: 'winnerwly',
}

// 非全局配置不需要
config.middleware = ['auth'];
// 非全局配置不需要
config.auth = {
  ignore: ['/user/login'], // 忽略不需要进行授权验证的接口
};
```

## 配置框架的扩展方法
```js
// /app/extend/context.js
// 如果原先就有其他方法可直接把getToken和checkToken两个方法补充在原有的后面
module.exports = {
  // 生成token
  getToken(value) {
    return this.app.jwt.sign(value, this.app.config.jwt.secret, { expiresIn: '1h' });
  },

  // 验证token
  checkToken(token) {
    return this.app.jwt.verify(token, this.app.config.jwt.secret);
  },
};
```

## 添加验证插件

```js
// /middleware/auth.js
// 检查token的操作全部在这里操作
module.exports = () => {
  return async function jwt(ctx, next) {
    const token = ctx.request.header.token;
    if (token) {
      try {
        ctx.state.user = ctx.checkToken(token);
        await next();
        return;
      } catch (error) {
        console.error('鉴权发生异常：', error);
        ctx.status = 401;
        ctx.body = {
          code: 401,
          message: '无效的身份认证',
        };
        return;
      }
    } else {
      ctx.status = 401;
      ctx.body = {
        code: 401,
        message: '没有token',
      };
      return;
    }
  };
};
```

## 登录接口获取token

```js
async login() {
  const { ctx, service } = this;
  const params = ctx.request.body;
  const result = await service.user.checkUserInfo({ username: params.username, password: params.password });
  if (result) {
    // 生成token同时把一些关键的用户信息存入token中，token是直接通过解密来处理有效性，服务端不会对token进行存储
    const token = ctx.getToken({ ...result.dataValues });
    this.responentSuccess({ data: { ...result.dataValues, token }, message: '登录成功' });
  } else {
    this.responentError({ message: '用户名或密码错误' });
  }
}
```



## 单独引入jwt进行调用

```js
const { router, controller, middleware } = app;
const jwt = middleware.auth(app.config.jwt);
router.post(`${baseUrl}/login`, controller.user.login);
router.get(`${baseUrl}/index`, jwt, controller.user.index);
```
